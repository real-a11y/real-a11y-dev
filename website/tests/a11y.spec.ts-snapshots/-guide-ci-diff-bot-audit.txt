  heading "3. Add the workflow" (level 2)
  button "Copy Code"
  code "# .github/workflows/a11y-diff.yml name: A11y diff on: pull_request: branches: [main] permissions: contents: read jobs: snapshot: name: Snapshot (${{ matrix.side }}) runs-on: ubuntu-latest strategy: matrix: side: [base, pr] steps: - uses: actions/checkout@v7 with: # base: the branch the PR targets. pr: empty, so the PR's merge commit. ref: ${{ matrix.side == 'base' && github.base_ref || '' }} - uses: actions/setup-node@v7 with: node-version: 22 cache: npm - run: npm ci - run: npx playwright install chromium --with-deps # Build and start your app. These are the steps to adapt. - run: npm run build - run: npx serve dist -l 3000 & - run: npx wait-on http://localhost:3000 --timeout 120000 - run: npx real-a11y snapshot -o ${{ matrix.side }}.json - uses: actions/upload-artifact@v7 with: name: ${{ matrix.side }}-snapshot path: ${{ matrix.side }}.json if-no-files-found: error diff: name: Diff & comment needs: snapshot runs-on: ubuntu-latest permissions: contents: read pull-requests: write # create and update the comment steps: # The base branch, not the PR: nothing from the PR runs with the write token. - uses: actions/checkout@v7 with: ref: ${{ github.base_ref }} - uses: actions/setup-node@v7 with: node-version: 22 cache: npm - run: npm ci # diff only reads two files: no browser needed - uses: actions/download-artifact@v8 with: pattern: "*-snapshot" merge-multiple: true # The full diff goes to the job log, and a shorter copy to the comment. - run: npx real-a11y diff base.json pr.json --explain --fail-on never - run: npx real-a11y diff base.json pr.json --format md --explain --fail-on never --max-pages 5 --max-lines 20 -o comment.md # A pull request from a fork gets a read-only token, so skip the comment there. - if: github.event.pull_request.head.repo.full_name == github.repository uses: actions/github-script@v9 with: script: | const fs = require("fs"); const MARKER = "<!-- real-a11y-diff -->"; const { owner, repo } = context.repo; const issue_number = context.issue.number; const run = `${context.serverUrl}/${owner}/${repo}/actions/runs/${context.runId}`; const body = `${MARKER}\n${fs.readFileSync("comment.md", "utf8")}\n[Full diff in the job log](${run})`; // Every page of comments: on a busy PR ours can be past the first, // and missing it posts a second comment on every push. const comments = await github.paginate(github.rest.issues.listComments, { owner, repo, issue_number, per_page: 100, }); // Match the bot AND a marker on the first line: a page's text can // quote the marker, and it ends up inside the comment body. const mine = comments.find( (c) => c.user?.login === "github-actions[bot]" && c.body?.startsWith(MARKER), ); if (mine) { await github.rest.issues.updateComment({ owner, repo, comment_id: mine.id, body }); } else { await github.rest.issues.createComment({ owner, repo, issue_number, body }); }"
  paragraph "Only the diff job can write to the PR, and it checks out the base branch, so the PR's install and build scripts never run with that permission. On a pull request from a fork, GitHub gives the workflow a read-only token whatever it asks for, so the comment step is skipped there and the diff is only in the job log. adds the plain-language lines under in the comment. and keep a large change readable, and the comment links to the job log for the rest. If a page shows content that changes on every build, such as a timestamp, add to both diff steps so it doesn't read as a change."
